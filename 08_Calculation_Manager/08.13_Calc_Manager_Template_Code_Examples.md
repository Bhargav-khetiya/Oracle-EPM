## 1. TEMPLATE: Standard P&L Calculation

```essbase
/* ===== TPL_Standard_PnL ===== */
SET LOCKBLOCK ON;
SET CALCPARALLEL 4;

FIX({RTP_Scenario}, "Working", &CurrYear)
  FIX(@RELATIVE("All_Hotels", 0))
    /* Revenue totals */
    "Total_Revenue" = "Room_Revenue" + "F&B_Revenue" + "Spa_Revenue";
    
    /* Expense totals */
    "Total_Expenses" = "Labor" + "Maintenance" + "Utilities";
    
    /* P&L */
    "Net_Income" = "Total_Revenue" - "Total_Expenses";
    
    /* Ratios (safe division) */
    IF("Total_Revenue" <> 0)
      "Profit_Margin" = "Net_Income" / "Total_Revenue" * 100;
      "Expense_Ratio" = "Total_Expenses" / "Total_Revenue" * 100;
    ELSE
      "Profit_Margin" = 0;
      "Expense_Ratio" = 0;
    ENDIF
  ENDFIX
  AGG("Entity");
ENDFIX

SET LOCKBLOCK OFF;
```

## 2. TEMPLATE: Budget Seed (Clear → Copy → Transform)

```essbase
/* ===== TPL_Budget_Seed ===== */
SET LOCKBLOCK ON;

/* Phase 1: Clear target */
FIX("Budget", "Working", &BudgetYear)
  CLEARDATA "Budget";
ENDFIX
CLEARBLOCK ALL;

/* Phase 2: Copy source to target */
SET CREATENONMISSINGBLK ON;
FIX("Working", @RELATIVE("All_Hotels", 0))
  DATACOPY "Actual" -> &PriorYear TO "Budget" -> &BudgetYear;
ENDFIX
SET CREATENONMISSINGBLK OFF;

/* Phase 3: Apply growth */
FIX("Budget", "Working", &BudgetYear, @RELATIVE("All_Hotels", 0))
  "Room_Revenue" = "Room_Revenue" * 1.05;
  "F&B_Revenue" = "F&B_Revenue" * 1.08;
  "Labor" = "Labor" * 1.03;
ENDFIX

SET LOCKBLOCK OFF;
```

## 3. TEMPLATE: Allocation (Revenue-Based)

```essbase
/* ===== TPL_Allocation ===== */
VAR v_TotalRevenue;
VAR v_CorpCost;
SET LOCKBLOCK ON;

FIX("All_Hotels", {RTP_Scenario}, "Working", &CurrYear, "YearTotal")
  v_TotalRevenue = "Total_Revenue";
ENDFIX

FIX("Corporate_HQ", {RTP_Scenario}, "Working", &CurrYear, "YearTotal")
  v_CorpCost = "Corp_Overhead";
ENDFIX

FIX({RTP_Scenario}, "Working", &CurrYear, @RELATIVE("All_Hotels", 0))
  IF(v_TotalRevenue <> 0)
    "Allocated_Corp" = v_CorpCost * ("Total_Revenue" / v_TotalRevenue);
  ENDIF
ENDFIX

SET LOCKBLOCK OFF;
```

## 4. TEMPLATE: Groovy Validation

```groovy
/* ===== TPL_Validation ===== */
/* RTPS: */
def errors = []

operation.grid.dataCellIterator("Account#Room_Revenue").each { cell ->
    if (cell.data < 0) {
        errors << "❌ Negative: ${cell.getMemberName('Entity')} / ${cell.getMemberName('Period')}"
    }
}

operation.grid.dataCellIterator("Account#Occupancy_Rate").each { cell ->
    if (!cell.isMissing() && cell.data > 100) {
        errors << "❌ Occ>100%: ${cell.getMemberName('Entity')}"
    }
}

if (errors) throwVetoException("Validation Failed:\n${errors.join('\n')}")
println "✅ Validation passed"
```
