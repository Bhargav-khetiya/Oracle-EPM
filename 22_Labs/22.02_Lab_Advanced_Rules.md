## 3. RULES TO BUILD

### Rule 1: CALC_Revenue_Totals

```essbase
/* Purpose: Sum revenue sub-accounts */
SET LOCKBLOCK ON;

FIX({RTP_Scenario}, "Working", &CurrYear)
  FIX(@RELATIVE("All_Hotels", 0))
    "Total_Revenue" = "Room_Revenue" + "F&B_Revenue" + "Spa_Revenue";
  ENDFIX
ENDFIX

SET LOCKBLOCK OFF;
```

### Rule 2: CALC_PnL_Complete

```essbase
/* Purpose: Full P&L with safe division */
SET LOCKBLOCK ON;
SET CALCPARALLEL 4;

FIX({RTP_Scenario}, "Working", &CurrYear)
  FIX(@RELATIVE("All_Hotels", 0))
    
    /* Revenue */
    "Total_Revenue" = "Room_Revenue" + "F&B_Revenue" + "Spa_Revenue";
    
    /* Expenses */
    "Total_Expenses" = "Labor" + "Maintenance" + "Utilities";
    
    /* P&L */
    "Net_Income" = "Total_Revenue" - "Total_Expenses";
    
    /* Ratios */
    IF("Total_Revenue" <> 0)
      "Profit_Margin" = "Net_Income" / "Total_Revenue" * 100;
      "Expense_Ratio" = "Total_Expenses" / "Total_Revenue" * 100;
    ELSE
      "Profit_Margin" = 0;
      "Expense_Ratio" = 0;
    ENDIF
    
  ENDFIX
  
  /* Aggregate */
  AGG("Entity");
  
ENDFIX

SET LOCKBLOCK OFF;
```

### Rule 3: COPY_Actual_to_Budget (Seed)

```essbase
/* Purpose: Clear budget, copy prior year actuals, apply growth */
SET LOCKBLOCK ON;

/* Phase 1: Clear */
FIX("Budget", "Working", &BudgetYear)
  CLEARDATA "Budget";
ENDFIX
CLEARBLOCK ALL;

/* Phase 2: Copy */
SET CREATENONMISSINGBLK ON;
FIX("Working", @RELATIVE("All_Hotels", 0))
  DATACOPY "Actual" -> &PriorYear TO "Budget" -> &BudgetYear;
ENDFIX
SET CREATENONMISSINGBLK OFF;

/* Phase 3: Apply 5% growth */
FIX("Budget", "Working", &BudgetYear, @RELATIVE("All_Hotels", 0))
  "Room_Revenue" = "Room_Revenue" * 1.05;
  "F&B_Revenue" = "F&B_Revenue" * 1.08;
  "Spa_Revenue" = "Spa_Revenue" * 1.10;
  "Labor" = "Labor" * 1.03;
  "Maintenance" = "Maintenance" * 1.03;
  "Utilities" = "Utilities" * 1.05;
ENDFIX

SET LOCKBLOCK OFF;
```

### Rule 4: ALLOC_Corporate_Cost

```essbase
/* Purpose: Allocate corporate overhead by revenue share */
VAR v_TotalRevenue;
VAR v_CorpOverhead;

SET LOCKBLOCK ON;

/* Get totals */
FIX("All_Hotels", {RTP_Scenario}, "Working", &CurrYear, "YearTotal")
  v_TotalRevenue = "Total_Revenue";
ENDFIX

FIX("Corporate_HQ", {RTP_Scenario}, "Working", &CurrYear, "YearTotal")
  v_CorpOverhead = "Total_Corp_Overhead";
ENDFIX

/* Allocate */
FIX({RTP_Scenario}, "Working", &CurrYear)
  FIX(@RELATIVE("All_Hotels", 0))
    IF(v_TotalRevenue <> 0)
      "Allocated_Corp_Cost" = v_CorpOverhead * 
        ("Total_Revenue" / v_TotalRevenue);
    ELSE
      "Allocated_Corp_Cost" = 0;
    ENDIF
  ENDFIX
ENDFIX

SET LOCKBLOCK OFF;
```

### Rule 5: VAL_Revenue_PreSave (Groovy)

```groovy
/* Purpose: Validate revenue before save — block negative values */
/* RTPS: */

def errors = []

// Check for negative revenue
operation.grid.dataCellIterator("Account#Room_Revenue").each { cell ->
    if (cell.data < 0) {
        String entity = cell.getMemberName("Entity")
        String period = cell.getMemberName("Period")
        errors << "❌ Negative Room Revenue: ${entity} / ${period} = ${cell.data}"
    }
}

operation.grid.dataCellIterator("Account#F&B_Revenue").each { cell ->
    if (cell.data < 0) {
        String entity = cell.getMemberName("Entity")
        String period = cell.getMemberName("Period")
        errors << "❌ Negative F&B Revenue: ${entity} / ${period} = ${cell.data}"
    }
}

// Block save if errors found
if (errors.size() > 0) {
    throwVetoException("Data Validation Failed:\n\n${errors.join('\n')}\n\nPlease correct and re-save.")
}

println "Validation passed ✅"
```
